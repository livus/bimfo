% \dogsticker{cx}{cy}{radius}: round sticker of the dog campaign on the bins, drawn here:
% a terrier on grass holding the "Nimm ein Sackerl fuer mein Gackerl" sign
\newcommand{\dogsticker}[3]{%
  \begin{scope}[shift={(#1,#2)}, yscale=-1, scale=#3]
    \shade[inner color=grasslight, outer color=grassdark] (0,0) circle (1.08);
    \fill[dogbrown] (-0.42,0.42) -- (-0.12,0.66) -- (-0.30,0.80) -- (-0.58,0.62) -- cycle;   % ears
    \fill[dogbrown] (0.42,0.42) -- (0.12,0.66) -- (0.30,0.80) -- (0.58,0.62) -- cycle;
    \fill[white] (0,0.22) ellipse (0.40 and 0.44);                                           % head
    \fill[white] (0,-0.08) ellipse (0.27 and 0.22);                                          % muzzle
    \fill[dogbrown] (-0.17,0.34) ellipse (0.20 and 0.25);                                    % patch
    \fill[black] (-0.15,0.27) circle (0.05) (0.15,0.27) circle (0.05);                       % eyes
    \fill[black] (0,0.02) ellipse (0.08 and 0.06);                                           % nose
    \fill[black!85] (-0.72,-0.36) rectangle (0.72,-0.70);                                    % sign
    \fill[white] (-0.36,-0.36) ellipse (0.11 and 0.08) (0.36,-0.36) ellipse (0.11 and 0.08); % paws
    \node[text=white, font=\bfseries\fontsize{1.05}{1.3}\selectfont, align=center, inner sep=0]
      at (0,-0.53) {NIMM EIN SACKERL\\F\"UR MEIN GACKERL};
    \draw[white, line width=0.15mm] (0,0) circle (1);
    \draw[cut, line width=0.08] (0,0) circle (1);
  \end{scope}}

% a filled decal of w x h, with 0.3 mm of colour past the cut line, and the cut line
% (y runs down the page)
\newcommand{\decal}[5]{% x, y (top left), w, h, colour
  \fill[#5] (#1-0.3, #2-0.3) rectangle (#1+#3+0.3, #2+#4+0.3);
  \draw[cut, line width=0.08] (#1, #2) rectangle (#1+#3, #2+#4);
}

% Box labels "Linie" / "Abfahrt in Minuten", for clear decal film: the model's own
% font (Liberation Sans Bold), scaled to the label as measured on the model:
% "Abfahrt in Minuten" 1.835 mm tall and 21.51 mm wide (21.38 mm of ink plus the
% sidebearings). OpenSCAD sets text about 4 % narrower than the font's own widths, so
% the two directions are scaled separately. One strip per face, as wide as
% the window (33.55 mm): "Linie" flush left, "Abfahrt in Minuten" flush right.
\newlength{\labh}
\newlength{\labw}
\newcommand{\labscale}{1}
\newcommand{\labxscale}{1}
\newcommand{\labeltext}[1]{\scalebox{\labxscale}[\labscale]{\fontsize{10}{12}\selectfont\bfseries #1}}
\newcommand{\labelstrip}[4]{% x, y (top left), strip height, baseline above the strip's bottom edge
  % crop marks outside the corners: on clear film a cut line would stay visible
  \foreach \cx/\dx in {#1/-1, #1+33.55/1} \foreach \cy/\dy in {#2/-1, #2+#3/1} {
    \draw[cut, line width=0.1] (\cx+\dx*0.6, \cy) -- (\cx+\dx*2.2, \cy);
    \draw[cut, line width=0.1] (\cx, \cy+\dy*0.6) -- (\cx, \cy+\dy*2.2);
  }
  \node[anchor=base west, inner sep=0] at (#1, #2+#3-#4) {\labeltext{Linie}};
  \node[anchor=base east, inner sep=0] at (#1+33.55, #2+#3-#4) {\labeltext{Abfahrt in Minuten}};
}
% The same on box grey, for plain sticker paper: cut on the line; the strip reaches from
% the window's top edge to an edge of the model (the face's chamfer, the back cover's edge).
\newcommand{\labelstripgrey}[4]{% x, y (top left), strip height, baseline above the strip's bottom edge
  \decal{#1}{#2}{33.55}{#3}{boxgrey}
  \node[anchor=base west, inner sep=0] at (#1, #2+#3-#4) {\labeltext{Linie}};
  \node[anchor=base east, inner sep=0] at (#1+33.55, #2+#3-#4) {\labeltext{Abfahrt in Minuten}};
}
